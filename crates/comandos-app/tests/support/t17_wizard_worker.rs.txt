struct Tmux { socket:PathBuf,replacement:PathBuf,reads:Rc<Cell<u32>>,swapped:Cell<bool>,recycled:Cell<bool>,active:Cell<&'static str>,scenario:&'static str,instance:Arc<AtomicBool>,scope:comandos_app::ui::snippets::Scope }
impl TmuxIo for Tmux {
    fn mode(&self)->RunMode{RunMode::Sandbox}
    fn socket(&self)->&Path{if self.swapped.get(){&self.replacement}else{&self.socket}}
    fn read(&self,args:&[&str])->Result<TmuxOut,TmuxError>{
        assert_eq!(&args[..3],["display-message","-p","-t"]);
        self.reads.set(self.reads.get()+1);
        let target=args[3];let fmt=args[4];
        let pane=if target=="=fixture:"{self.active.get()}else{target};
        assert!(["%2","%9","%3"].contains(&pane));
        let stdout=if fmt.contains("#{pid}") {
            format!("{}|$1|100|fixture|{pane}",if self.recycled.get(){99}else{21})
        } else if fmt=="#{session_name}\n#{pane_id}" {
            if self.scenario=="active_changed"{self.active.set("%3");}
            format!("fixture\n{pane}")
        } else {
            assert_eq!(fmt,"#{session_name}\n#{pane_id}\n#{pane_current_path}");
            match self.scenario {
                "recycled"=>self.recycled.set(true),
                "socket_changed"=>self.swapped.set(true),
                "cancelled"=>self.instance.store(true,Ordering::Release),
                "generation_changed"=>self.scope.advance(),
                "active_changed"=>self.active.set("%3"),
                _=>{}
            }
            format!("fixture\n{pane}\ncwd{}",&pane[1..])
        };
        Ok(TmuxOut{code:0,stdout,stderr:String::new()})
    }
    fn mutate(&self,_:&[&str],_:Option<&[u8]>)->Result<TmuxOut,TmuxError>{panic!("wizard must not mutate tmux")}
}
