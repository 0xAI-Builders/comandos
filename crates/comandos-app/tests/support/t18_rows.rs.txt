}
fn main(){let prev=Button::new("tab-cycle-prev");let next=Button::new("tab-cycle-next");let rows=Button::new("tabs-layout-rows");let posts=Rc::new(Cell::new(0));let app=Rc::new(App{tab_layout:Layout{rows:Cell::new(false),start:Row(vec![prev.clone()]),actions:Row(vec![rows.clone(),next.clone()])},applied_theme:RefCell::new(Some(Theme{values:json!({"brand":"owned-brand","dim":"owned-dim"})})),english:true,t18:Owned{scope:Scope::default()},closed:Arc::new(AtomicBool::new(false)),dash:Dash(posts.clone()),jobs:Jobs,ready:Cell::new(true)});
app.apply_tabs_layout("rows",false);assert_eq!(posts.get(),0);assert!(app.tab_layout.rows.get());assert!(!prev.visible.get()&&!next.visible.get());assert_eq!(rows.image.borrow().as_str(),"owned-brand");assert_eq!(rows.tip.borrow().as_str(),"One row");

app.apply_tabs_layout("rows",true);assert_eq!(posts.get(),1);app.ready.set(false);app.apply_tabs_layout("row",true);assert_eq!(posts.get(),1);app.ready.set(true);app.closed.store(true,Ordering::Release);app.apply_tabs_layout("row",true);assert_eq!(posts.get(),1);println!("{}",json!({"posts":1,"cases":5}));}
