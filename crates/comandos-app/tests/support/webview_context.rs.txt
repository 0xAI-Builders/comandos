use std::{cell::RefCell, collections::BTreeMap, path::Path, rc::Rc};
fn fixture_dir() -> &'static Path {
    static DIR: std::sync::OnceLock<std::path::PathBuf> = std::sync::OnceLock::new();
    DIR.get_or_init(|| std::env::temp_dir().join(format!("comandos-webview-auth-{}", std::process::id()))).as_path()
}

#[derive(Clone, Default)]
struct WebContext(Rc<RefCell<BTreeMap<String, String>>>);
impl WebContext {
    fn with_website_data_manager(_: &webkit2gtk::WebsiteDataManager) -> Self { Self::default() }
}
mod webkit2gtk {
    #[derive(Clone,Debug)] pub enum UserContentInjectedFrames { TopFrame }
    #[derive(Clone,Debug)] pub enum UserScriptInjectionTime { Start }
    #[derive(Clone)] pub struct UserScript(pub serde_json::Value);
    impl UserScript { pub fn new(source:&str, frames:UserContentInjectedFrames, time:UserScriptInjectionTime, allow:&[&str], _: &[&str])->Self { Self(serde_json::json!({"source":source,"frames":format!("{frames:?}"),"time":format!("{time:?}"),"allow":allow})) } }
    pub struct WebsiteDataManager;
    impl WebsiteDataManager {
        pub fn new_ephemeral() -> Self { Self }
        pub fn builder() -> Self { Self }
        pub fn base_data_directory(self, _: &str) -> Self { self }
        pub fn base_cache_directory(self, _: &str) -> Self { self }
        pub fn build(self) -> Self { self }
    }
    #[derive(Clone, Default)]
    pub struct UserContentManager(pub std::rc::Rc<std::cell::RefCell<Vec<String>>>, pub std::rc::Rc<std::cell::RefCell<Vec<UserScript>>>);
    impl UserContentManager {
        pub fn new() -> Self { Self::default() }
        pub fn register_script_message_handler(&self, name: &str) -> bool {
            self.0.borrow_mut().push(name.into()); true
        }
        pub fn add_script(&self, script:&UserScript) { self.1.borrow_mut().push(script.clone()); }
    }
}
#[derive(Clone)]
struct WebView { context: WebContext, manager: webkit2gtk::UserContentManager }
struct Builder { context: Option<WebContext>, manager: Option<webkit2gtk::UserContentManager> }
impl WebView {
    fn builder() -> Builder { Builder { context: None, manager: None } }
    fn context(&self) -> Option<WebContext> { Some(self.context.clone()) }
}
impl Builder {
    fn web_context(mut self, context: &WebContext) -> Self { self.context = Some(context.clone()); self }
    fn user_content_manager(mut self, manager: &webkit2gtk::UserContentManager) -> Self { self.manager = Some(manager.clone()); self }
    fn build(self) -> WebView { WebView { context: self.context.unwrap(), manager: self.manager.unwrap() } }
}

fn main() {
    use std::os::unix::fs::DirBuilderExt;
    std::fs::DirBuilder::new().mode(0o700).create(fixture_dir()).unwrap();
    std::fs::write(fixture_dir().join("dash-token"), "private-\"fixture\\token\n").unwrap();
    let main = create(&AppConfig(RunMode::Live), "centro", None).unwrap();
    main.context.0.borrow_mut().insert("cc-center-panel".into(), "settings".into());
    let modal = create(&AppConfig(RunMode::Live), "centro", Some(&main)).unwrap();
    let other = create(&AppConfig(RunMode::Sandbox), "centro", None).unwrap();
    let shadow = create(&AppConfig(RunMode::Shadow), "centro", None).unwrap();
    std::fs::remove_file(fixture_dir().join("dash-token")).unwrap();
    let missing = create(&AppConfig(RunMode::Live), "centro", None).unwrap();
    std::fs::remove_dir(fixture_dir()).unwrap();
    println!("{{\"shared_context\":{},\"request_visible\":{},\"separate_handlers\":{},\"profiles_isolated\":{},\"authenticated_views\":{},\"shadow_scripts\":{},\"missing_token_scripts\":{},\"script\":{}}}",
        Rc::ptr_eq(&main.context.0, &modal.context.0),
        modal.context.0.borrow().get("cc-center-panel").is_some_and(|v| v == "settings"),
        !Rc::ptr_eq(&main.manager.0, &modal.manager.0),
        !Rc::ptr_eq(&main.context.0, &other.context.0),
        [&main,&modal,&other].iter().filter(|v|v.manager.1.borrow().len()==1).count(),
        shadow.manager.1.borrow().len(),missing.manager.1.borrow().len(),
        main.manager.1.borrow().first().map(|s|s.0.clone()).unwrap_or(serde_json::Value::Null));
}
